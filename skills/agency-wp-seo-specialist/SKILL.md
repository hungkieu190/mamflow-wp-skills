## 📋 Lightweight JSON-LD Breadcrumbs for WordPress
```php
<?php
defined( 'ABSPATH' ) || exit;

add_action( 'wp_head', 'mamflow_render_schema_breadcrumbs' );
function mamflow_render_schema_breadcrumbs(): void {
    if ( is_front_page() ) {
        return;
    }

    $schema = array(
        '@context'        => 'https://schema.org',
        '@type'           => 'BreadcrumbList',
        'itemListElement' => array(
            array(
                '@type'    => 'ListItem',
                'position' => 1,
                'name'     => get_bloginfo( 'name' ),
                'item'     => home_url( '/' ),
            ),
        ),
    );

    if ( is_single() || is_page() ) {
        $schema['itemListElement'][] = array(
            '@type'    => 'ListItem',
            'position' => 2,
            'name'     => get_the_title(),
            'item'     => get_permalink(),
        );
    }

    echo '<script type="application/ld+json">' . wp_json_encode( $schema, JSON_UNESCAPED_SLASHES | JSON_UNESCAPED_UNICODE ) . '</script>' . "\n";
}
```
